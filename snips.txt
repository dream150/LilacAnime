   195	        current: FloatArray,
   196	        template: OfflineOpEdFingerprintStore.Template,
   197	        duration: Double,
   198	        status: (String) -> Unit
   199	    ): List<ChapterSkipSegment> {
   200	        val out = ArrayList<ChapterSkipSegment>()

   203	            val end = it.endSeconds.coerceIn(it.startSeconds, duration)
   204	            out += ChapterSkipSegment("op", it.startSeconds.coerceIn(0.0, duration), end, duration)

   219	    private fun trainFingerprintTemplates(
   220	        fps: List<Pair<Episode, FloatArray>>,
   221	        status: (String) -> Unit
   222	    ): Pair<FloatArray?, FloatArray?>? {
   223	            // Build real audio templates from complete-episode repetition.
   224	            // IMPORTANT: no episode timestamp is averaged or persisted here.  Every
   225	            // candidate is an actual slice of audio fingerprint data.  We validate
   226	            // that the same audio occurs in every training episode, then cluster
   227	            // equivalent fingerprints so OP/ED are selected by recurrence rather
   228	            // than by their absolute position in any one episode.
   229	            data class Occurrence(
   230	                val episodeId: String,
   231	                val episodeNumber: Int,
   232	                val startSeconds: Double,
   233	                val endSeconds: Double,
   234	                val score: Double
   235	            )
   236	            data class TrainingCandidate(
   237	                val fingerprint: FloatArray,
   238	                val occurrences: MutableList<Occurrence>,
   239	                val quality: Double
   240	            )
   241	            data class TrainingCluster(
   242	                var fingerprint: FloatArray,
   243	                val occurrences: MutableList<Occurrence>,
   244	                var quality: Double
   245	            )
   246	
   247	            val candidates = ArrayList<TrainingCandidate>()
   248	            for (aIndex in fps.indices) {
   249	                for (bIndex in aIndex + 1 until fps.size) {
   250	                    val found = findTopRepeatedRegions(fps[aIndex].second, fps[bIndex].second, 12)
   251	                    status(
   252	                        "FINGERPRINT_TRAIN_PAIR a=${fps[aIndex].first.number} " +
   253	                            "b=${fps[bIndex].first.number} candidates=${found.size}"
   254	                    )
   255	
   256	                    for (seed in found) {
   257	                        val templateCandidate = fps[aIndex].second
   258	                            .sliceFingerprint(seed.startSeconds, seed.endSeconds)
   259	                        if (templateCandidate.size < FINGERPRINT_BINS * 300) continue // at least 30 seconds
   260	
   261	                        val occurrences = ArrayList<Occurrence>()
   262	                        var qualitySum = 0.0
   263	                        var valid = true
   264	
   265	                        // Search the ENTIRE fingerprint of every training episode for
   266	                        // this exact audio template.  The location is only an
   267	                        // occurrence discovered during training; it is never stored.
   268	                        for ((episode, fingerprint) in fps) {
   269	                            val match = scoreTemplateAgainstCurrent(
   270	                                fingerprint,
   271	                                templateCandidate,
   272	                                "TRAIN_OCCURRENCE_${episode.number}",
   273	                                ::status
   274	                            )
   275	                            if (match == null || match.score < 0.82 || match.endSeconds <= match.startSeconds) {
   276	                                valid = false
   277	                                break
   278	                            }
   279	                            occurrences += Occurrence(
   280	                                episode.id,
   281	                                episode.number,
   282	                                match.startSeconds,
   283	                                match.endSeconds,
   284	                                match.score
   285	                            )
   286	                            qualitySum += match.score
   287	                        }
   288	
   289	                        if (valid && occurrences.size == fps.size) {
   290	                            candidates += TrainingCandidate(
   291	                                templateCandidate,
   292	                                occurrences,
   293	                                qualitySum / occurrences.size.coerceAtLeast(1)
   294	                            )
   295	                        }
   296	                    }
   297	                }
   298	            }
   299	
   300	            if (candidates.isEmpty()) {
   301	                status("FINGERPRINT_TRAINING_FAILED no_common_audio_templates")
   302	                return null
   303	            }
   304	
   305	            // Merge the same OP/ED audio found from different episode pairs.
   306	            // Clustering is based on the fingerprint samples themselves, never on
   307	            // start/end timestamps.  This is what allows episode 1 to have the OP
   308	            // at a completely different position from the other episodes.
   309	            val clusters = ArrayList<TrainingCluster>()
   310	            for (candidate in candidates.sortedByDescending { it.quality }) {
   311	                var matchedCluster: TrainingCluster? = null
   312	                for (cluster in clusters) {
   313	                    val minLength = minOf(candidate.fingerprint.size, cluster.fingerprint.size)
   314	                    val maxLength = maxOf(candidate.fingerprint.size, cluster.fingerprint.size)
   315	                    if (minLength < 300 || minLength.toDouble() / maxLength.toDouble() < 0.60) continue
   316	                    val similarity = cosine(
   317	                        candidate.fingerprint, 0,
   318	                        cluster.fingerprint, 0,
   319	                        minLength
   320	                    )
   321	                    if (similarity >= 0.90) {
   322	                        matchedCluster = cluster
   323	                        break
   324	                    }
   325	                }
   326	
   327	                if (matchedCluster == null) {
   328	                    clusters += TrainingCluster(
   329	                        candidate.fingerprint.copyOf(),
   330	                        ArrayList(candidate.occurrences),
   331	                        candidate.quality
   332	                    )
   333	                } else {
   334	                    // Keep the best-quality actual audio fingerprint as the
   335	                    // representative; merge only the discovered occurrences.
   336	                    for (occurrence in candidate.occurrences) {
   337	                        val old = matchedCluster.occurrences.indexOfFirst { it.episodeId == occurrence.episodeId }
   338	                        if (old < 0) {
   339	                            matchedCluster.occurrences += occurrence
   340	                        } else if (occurrence.score > matchedCluster.occurrences[old].score) {
   430	                    opCluster = second
   431	                    edCluster = first
   432	                }
   433	                status("FINGERPRINT_TRAIN_ORDER_FALLBACK")
   434	            }
   435	
   436	            val opTemplate = opCluster?.fingerprint?.copyOf()
   437	            val edTemplate = edCluster?.fingerprint?.copyOf()
   438	            status(
   439	                "FINGERPRINT_TEMPLATES_SELECTED " +
   440	                    "opSeconds=${opTemplate?.let { fingerprintFrames(it) * FINGERPRINT_FRAME_SECONDS } ?: 0.0} " +
   441	                    "edSeconds=${edTemplate?.let { fingerprintFrames(it) * FINGERPRINT_FRAME_SECONDS } ?: 0.0} " +
   442	                    "clusters=${recurring.size}"
   443	            )
   444	            opCluster?.let { cluster ->
   445	                status("FINGERPRINT_OP_OCCURRENCES " + cluster.occurrences.joinToString(";") {
   446	                    "ep=${it.episodeNumber}@${it.startSeconds}-${it.endSeconds},score=${it.score}"
   447	                })
   448	            }
   449	            edCluster?.let { cluster ->
   450	                status("FINGERPRINT_ED_OCCURRENCES " + cluster.occurrences.joinToString(";") {
   451	                    "ep=${it.episodeNumber}@${it.startSeconds}-${it.endSeconds},score=${it.score}"
   452	                })
   453	            }
   454	
   455	
   456	            return opTemplate to edTemplate
   457	    }

  1080	    /**
  1081	     * Finds long, contiguous repeated audio regions between two complete episodes.
  1082	     * A 20-second seed is only the starting point.  The candidate is extended while
  1083	     * adjacent 5-second blocks continue to match, so the persisted template is the
  1084	     * actual repeated audio region rather than a fixed analysis window.
  1085	     */
  1086	    private fun findTopRepeatedRegions(a: FloatArray, b: FloatArray, maxResults: Int): List<Match> {
  1087	        // Work in fingerprint frames, not flattened mel-bin samples.
  1088	        // A seed is a 20-second sequence (200 x 0.1s frames).  The score is the
  1089	        // average cosine similarity of corresponding 32-bin frames.  This makes
  1090	        // the meaning of the score independent of FINGERPRINT_BINS and avoids
  1091	        // treating a flattened 454,752-element array as a single frame sequence.
  1092	        val seedBlock = 200 // 20 seconds
  1093	        val extensionBlock = 50 // 5 seconds
  1094	        val step = 20 // 2 seconds during discovery
  1095	        val aFrames = fingerprintFrames(a)
  1096	        val bFrames = fingerprintFrames(b)
  1097	        if (aFrames < seedBlock || bFrames < seedBlock) return emptyList()
  1098	
  1099	        val seeds = ArrayList<Match>()
  1100	        var globalBest = -1.0
  1101	        var globalBestA = -1
  1102	        var globalBestB = -1
  1103	
  1104	        var ai = 0
  1105	        while (ai + seedBlock <= aFrames) {
  1106	            var bestBi = -1
  1107	            var bestScore = -1.0
  1108	            var bi = 0
  1109	            while (bi + seedBlock <= bFrames) {
  1110	                val score = sequenceCosineFrames(a, ai, b, bi, seedBlock)
  1111	                if (score > bestScore) {
  1112	                    bestScore = score
  1113	                    bestBi = bi
  1114	                }
  1115	                bi += step
  1116	            }
  1117	
  1118	            if (bestScore > globalBest) {
  1119	                globalBest = bestScore
  1120	                globalBestA = ai
  1121	                globalBestB = bestBi
  1122	            }
  1123	
  1124	            // Do not require an unrealistically high score at the discovery stage.
  1125	            // The later full-template validation decides whether the repeated audio
  1126	            // is strong enough to become a persisted template.
  1127	            if (bestBi >= 0 && bestScore >= 0.55) {
  1128	                var leftA = ai
  1129	                var leftB = bestBi
  1130	                var rightA = ai + seedBlock
  1131	                var rightB = bestBi + seedBlock
  1132	                var goodBlocks = 1
  1133	                var totalBlocks = 1
  1134	                var extensionScoreSum = bestScore
  1135	
  1136	                while (leftA - extensionBlock >= 0 && leftB - extensionBlock >= 0) {
  1137	                    val score = sequenceCosineFrames(a, leftA - extensionBlock, b, leftB - extensionBlock, extensionBlock)
  1138	                    totalBlocks++
  1139	                    if (score < 0.55) break
  1140	                    goodBlocks++
  1141	                    extensionScoreSum += score
  1142	                    leftA -= extensionBlock
  1143	                    leftB -= extensionBlock
  1144	                }
  1145	
  1146	                while (rightA + extensionBlock <= aFrames && rightB + extensionBlock <= bFrames) {
  1147	                    val score = sequenceCosineFrames(a, rightA, b, rightB, extensionBlock)
  1148	                    totalBlocks++
  1149	                    if (score < 0.55) break
  1150	                    goodBlocks++
  1151	                    extensionScoreSum += score
  1152	                    rightA += extensionBlock
  1153	                    rightB += extensionBlock
  1154	                }
  1155	
  1156	                val continuity = goodBlocks.toDouble() / totalBlocks.coerceAtLeast(1)
  1157	                val average = extensionScoreSum / totalBlocks.coerceAtLeast(1)
  1158	                val score = bestScore * 0.45 + average * 0.35 + continuity * 0.20
  1159	                val duration = (rightA - leftA) * FINGERPRINT_FRAME_SECONDS
  1160	                if (score >= 0.60 && continuity >= 0.50 && duration >= 40.0) {
  1161	                    seeds += Match(
  1162	                        leftA * FINGERPRINT_FRAME_SECONDS,
  1163	                        rightA * FINGERPRINT_FRAME_SECONDS,
  1164	                        score
  1165	                    )
  1166	                }
  1167	            }
  1168	            ai += step
  1169	        }
  1170	
  1171	        // Always expose the best pair score.  This is diagnostic information, not
  1172	        // a persisted template: it lets us distinguish "no similarity" from a
  1173	        // candidate being rejected by a later validation rule.
  1174	        Log.d(
  1175	            TAG,
  1176	            "FINGERPRINT_PAIR_BEST score=$globalBest " +
  1177	                "aStart=${globalBestA * FINGERPRINT_FRAME_SECONDS} " +
  1178	                "bStart=${globalBestB * FINGERPRINT_FRAME_SECONDS}"
  1179	        )
  1180	
  1181	        val selected = ArrayList<Match>()
  1182	        for (candidate in seeds.sortedByDescending { it.score }) {
  1183	            if (selected.none {
  1184	                    kotlin.math.abs(it.startSeconds - candidate.startSeconds) < 30.0 ||
  1185	                        rangesOverlap(it.startSeconds, it.endSeconds, candidate.startSeconds, candidate.endSeconds)
  1186	                }) {
  1187	                selected += candidate
  1188	                if (selected.size >= maxResults) break
  1189	            }
  1190	        }
  1191	        return selected.sortedBy { it.startSeconds }
  1192	    }

  1195	        aStart < bEnd && bStart < aEnd

  1198	        val frameCount = size / FINGERPRINT_BINS

  1200	        val eFrame = (endSeconds / FINGERPRINT_FRAME_SECONDS).toInt().coerceIn(sFrame, frameCount)

  1209	    private fun sequenceCosineFrames(
  1210	        a: FloatArray,

  1241	     * template into an arbitrary 90/400-second chapter because adjacent audio also
  1242	     * happens to correlate.
  1243	     */
  1244	    private fun scoreTemplateAgainstCurrent(
  1245	        current: FloatArray,
  1246	        template: FloatArray,
  1247	        label: String,
  1248	        status: (String) -> Unit,
  1249	        minStartSeconds: Double = 0.0
  1250	    ): Match? {
  1251	        if (template.size < FINGERPRINT_BINS * 10 || current.size < template.size) return null
  1252	
  1253	        val scanStep = 10 // 1 second coarse scan in 0.1s frames; final score uses the whole template.
  1254	        val currentFrames = fingerprintFrames(current)
  1255	        val templateFrames = fingerprintFrames(template)
  1256	        val minStartFrame = kotlin.math.ceil(
  1257	            minStartSeconds / FINGERPRINT_FRAME_SECONDS
  1258	        ).toInt().coerceIn(0, currentFrames)
  1259	        status(
  1260	            "FINGERPRINT_${label}_SEARCH_START seconds=$minStartSeconds frame=$minStartFrame " +
  1261	                "duration=${currentFrames * FINGERPRINT_FRAME_SECONDS}"
  1262	        )
  1263	        var bestStart = -1
  1264	        var bestCosine = -1.0
  1265	        var i = minStartFrame
  1266	        while (i + templateFrames <= currentFrames) {
  1267	            val c = sequenceCosineFrames(current, i, template, 0, templateFrames)
  1268	            if (c > bestCosine) {
  1269	                bestCosine = c
  1270	                bestStart = i
  1271	            }
  1272	            i += scanStep
  1273	        }
  1274	
  1275	        if (bestStart < 0 || bestCosine < 0.65) {
  1276	            status("FINGERPRINT_${label}_REJECT cosine=$bestCosine minStartSeconds=$minStartSeconds")
  1277	            return null
  1278	        }
  1279	
  1280	        // Validate the same candidate over the entire template in 5-second blocks.
  1281	        // This prevents one short high-correlation patch from determining the match.
  1282	        val block = 50 // 5 seconds in fingerprint frames
  1283	        var good = 0
  1284	        var total = 0
  1285	        var sum = 0.0
  1286	        var q = 0
  1287	        while (q + block <= templateFrames) {
  1288	            val c = sequenceCosineFrames(current, bestStart + q, template, q, block)
  1289	            sum += c
  1290	            total++
  1291	            if (c >= 0.55) good++
  1292	            q += block
  1293	        }
  1294	        if (q < templateFrames) {
  1295	            val remaining = templateFrames - q
  1296	            if (remaining >= 10) {
  1297	                val c = sequenceCosineFrames(current, bestStart + q, template, q, remaining)
  1298	                sum += c
  1299	                total++
  1300	                if (c >= 0.55) good++
  1301	            }
  1302	        }
  1303	
  1304	        val continuity = good.toDouble() / total.coerceAtLeast(1)
  1305	        val average = sum / total.coerceAtLeast(1)
  1306	        val score = bestCosine * 0.40 + continuity * 0.35 + average * 0.25
  1307	        status(
  1308	            "FINGERPRINT_${label}_SCORE cosine=$bestCosine " +
  1309	                "continuity=$continuity average=$average blocks=$good/$total score=$score"
  1310	        )
  1311	
  1312	        if (score < 0.65 || continuity < 0.60) {
  1313	            status("FINGERPRINT_${label}_REJECT reason=continuity score=$score continuity=$continuity")
  1314	            return null
  1315	        }
  1316	
  1317	        // The persisted fingerprint length defines the chapter length exactly.
  1318	        val start = bestStart * FINGERPRINT_FRAME_SECONDS
  1319	        val end = (bestStart + fingerprintFrames(template)) * FINGERPRINT_FRAME_SECONDS
  1320	        status(
  1321	            "FINGERPRINT_${label}_MATCH start=$start end=$end " +
  1322	                "score=$score templateSeconds=${fingerprintFrames(template) * FINGERPRINT_FRAME_SECONDS}"
  1323	        )
  1324	        return Match(start, end, score)
  1325	    }
